    11	import {

    40	  'application/msword', // .doc
    41	  'text/plain',
    42	  'text/markdown',

    44	  'image/jpeg',
    45	  'image/png',
    46	  'image/webp'

    48

    50	
    51	export const StudentLogsTab: React.FC = () => {
    52	  const [activeSectionView, setActiveSectionView] = useState<'all' | 'notes' | 'skills' | 'project-docs' | 'learning-process'>('all');
    53	
    54	  const [meetingNotes, setMeetingNotes] = useState<StudentMeetingNote[]>([]);
    55	  const [skillUpdates, setSkillUpdates] = useState<StudentSkillUpdate[]>([]);
    56	  const [projectDocs, setProjectDocs] = useState<StudentProjectDocument[]>([]);
    57	  const [learningNotes, setLearningNotes] = useState<StudentLearningProcessNote[]>([]);
    58	  const [loading, setLoading] = useState(true);
    59	
    60	  // Modal visibility states
    61	  const [showNoteModal, setShowNoteModal] = useState(false);
    62	  const [showSkillModal, setShowSkillModal] = useState(false);
    63	  const [showDocModal, setShowDocModal] = useState(false);
    64	  const [showLearningModal, setShowLearningModal] = useState(false);
    65	
    66	  // Preview modal states
    67	  const [previewDoc, setPreviewDoc] = useState<StudentProjectDocument | null>(null);
    68	  const [previewUrl, setPreviewUrl] = useState<string>('');
    69	  const [isPdf, setIsPdf] = useState(false);
    70	  const [isImage, setIsImage] = useState(false);
    71	  const [loadingPreview, setLoadingPreview] = useState(false);
    72	  const [previewError, setPreviewError] = useState<string | null>(null);
    73	
    74	  // Submission / form states
    75	  const [submitting, setSubmitting] = useState(false);
    76	  const [errorMsg, setErrorMsg] = useState<string | null>(null);
    77	  const [successMsg, setSuccessMsg] = useState<string | null>(null);
    78	
    79	  // Form states
    80	  const [noteForm, setNoteForm] = useState({ meeting_topic: '', notes: '', learnt: '', feedback: '' });
    81	  const [skillForm, setSkillForm] = useState({ skill_name: '', description: '' });
    82	  const [docForm, setDocForm] = useState({ title: '', description: '' });
    83	  const [selectedDocFile, setSelectedDocFile] = useState<File | null>(null);
    84	  const [learningForm, setLearningForm] = useState({ title: '', notes: '', challenges: '', key_milestones: '' });
    85	
    86	  const fileInputRef = useRef<HTMLInputElement>(null);
    87	
    88	  useEffect(() => {
    89	    fetchData();
    90	  }, []);
    91	
    92	  const fetchData = async () => {
    93	    setLoading(true);
    94	    const { data: { session } } = await supabase.auth.getSession();
    95	    if (!session) {
    96	      setLoading(false);
    97	      return;
    98	    }
    99	
   100	    try {
   101	      const [notesRes, skillsRes, docsRes, learnRes] = await Promise.all([
   102	        supabase.from('student_meeting_notes').select('*').eq('user_id', session.user.id).order('created_at', { ascending: false }),
   103	        supabase.from('student_skills_updates').select('*').eq('user_id', session.user.id).order('created_at', { ascending: false }),
   104	        supabase.from('student_project_documents').select('*').eq('user_id', session.user.id).order('created_at', { ascending: false }),
   105	        supabase.from('student_learning_process_notes').select('*').eq('user_id', session.user.id).order('created_at', { ascending: false }),
   106	      ]);
   107	
   108	      if (notesRes.data) setMeetingNotes(notesRes.data as StudentMeetingNote[]);
   109	      if (skillsRes.data) setSkillUpdates(skillsRes.data as StudentSkillUpdate[]);
   110	      if (docsRes.data) setProjectDocs(docsRes.data as StudentProjectDocument[]);
   111	      if (learnRes.data) setLearningNotes(learnRes.data as StudentLearningProcessNote[]);
   112	    } catch (e) {
   113	      console.warn('Failed to fetch logs:', e);
   114	    } finally {
   115	      setLoading(false);
   116	    }
   117	  };
   118	
   119	  // Preview resolution effect
   120	  useEffect(() => {
   121	    if (!previewDoc) {
   122	      setPreviewUrl('');
   123	      setPreviewError(null);
   124	      return;
   125	    }
   126	
   127	    let isMounted = true;
   128	    const fetchPreview = async () => {
   129	      setLoadingPreview(true);
   130	      setPreviewError(null);
   131	
   132	      try {
   133	        const res = await resolveDocumentPreview(
   134	          previewDoc.file_path,
   135	          previewDoc.file_name,
   136	          previewDoc.file_type
   137	        );
   138	
   139	        if (!isMounted) return;
   140	
   141	        if (res.url) {
   142	          setPreviewUrl(res.url);
   143	          setIsPdf(res.isPdf);
   144	          setIsImage(res.isImage);
   145	        } else {
   146	          setPreviewError(res.error || 'Preview could not be loaded.');
   147	        }
   148	      } catch (err: any) {
   149	        if (!isMounted) return;
   150	        setPreviewError(err.message || 'Failed to render document preview.');
   151	      } finally {
   152	        if (isMounted) setLoadingPreview(false);
   153	      }
   154	    };
   155	
   156	    fetchPreview();
   157	
   158	    return () => {
   159	      isMounted = false;
   160	    };
   161	  }, [previewDoc]);
   162	
   163	  const handleSaveNote = async () => {
   164	    setErrorMsg(null);
   165	    if (!noteForm.meeting_topic.trim() || !noteForm.notes.trim() || !noteForm.learnt.trim()) {
   166	      setErrorMsg('Please fill out all required fields.');
   167	      return;
   168	    }
   169	
   170	    setSubmitting(true);
   171	    const { data: { session } } = await supabase.auth.getSession();
   172	    if (!session) return;
   173	
   174	    try {
   175	      const { data, error } = await supabase
   176	        .from('student_meeting_notes')
   177	        .insert([{ user_id: session.user.id, ...noteForm }])
   178	        .select()
   179	        .single();
   180	
   181	      if (error) {
   182	        if (error.code === '42P01') throw new Error("Table 'student_meeting_notes' does not exist yet. Please run the SQL schema script in your Supabase dashboard.");
   183	        throw error;
   184	      }
   185	
   186	      setMeetingNotes([data, ...meetingNotes]);
   187	      setShowNoteModal(false);
   188	      setNoteForm({ meeting_topic: '', notes: '', learnt: '', feedback: '' });
   189	      setSuccessMsg('Meeting note logged successfully!');
   190	      setTimeout(() => setSuccessMsg(null), 3000);
   191	    } catch (err: any) {
   192	      setErrorMsg(err.message || 'Failed to save meeting note');
   193	    } finally {
   194	      setSubmitting(false);
   195	    }
   196	  };
   197	
   198	  const handleSaveSkill = async () => {
   199	    setErrorMsg(null);
   200	    if (!skillForm.skill_name.trim() || !skillForm.description.trim()) {
   201	      setErrorMsg('Please fill out all required fields.');
   202	      return;
   203	    }
   204	
   205	    setSubmitting(true);
   206	    const { data: { session } } = await supabase.auth.getSession();
   207	    if (!session) return;
   208	
   209	    try {
   210	      const { data, error } = await supabase
   211	        .from('student_skills_updates')
   212	        .insert([{ user_id: session.user.id, ...skillForm }])
   213	        .select()
   214	        .single();
   215	
   216	      if (error) {
   217	        if (error.code === '42P01') throw new Error("Table 'student_skills_updates' does not exist yet. Please run the SQL schema script in your Supabase dashboard.");
   218	        throw error;
   219	      }
   220	
   221	      setSkillUpdates([data, ...skillUpdates]);
   222	      setShowSkillModal(false);
   223	      setSkillForm({ skill_name: '', description: '' });
   224	      setSuccessMsg('Skill update logged successfully!');
   225	      setTimeout(() => setSuccessMsg(null), 3000);
   226	    } catch (err: any) {
   227	      setErrorMsg(err.message || 'Failed to save skill update');
   228	    } finally {
   229	      setSubmitting(false);
   230	    }
   231	  };
   232	
   233	  const handleFileSelection = (e: React.ChangeEvent<HTMLInputElement>) => {
   234	    setErrorMsg(null);
   235	    if (e.target.files && e.target.files[0]) {
   236	      const file = e.target.files[0];
   237	      const lowerName = file.name.toLowerCase();
   238	
   239	      const isValidExt = [
   240	        '.pdf',
   241	        '.docx',
   242	        '.doc',
   243	        '.txt',
   244	        '.md',
   245	        '.rtf',
   246	        '.jpg',
   247	        '.jpeg',
   248	        '.png'
   249	      ].some((ext) => lowerName.endsWith(ext));
   250	
   251	      const isPermittedType = ALLOWED_DOC_TYPES.includes(file.type) || file.type === 'application/octet-stream' || !file.type;
   252	
   253	      if (!isValidExt && !isPermittedType) {
   254	        setErrorMsg('Invalid file format. Please upload a Word document, PDF, or text document.');
   255	        setSelectedDocFile(null);
   256	        return;
   257	      }
   258	
   259	      if (file.size > MAX_DOC_SIZE) {
   260	        setErrorMsg(`File size (${(file.size / (1024 * 1024)).toFixed(1)} MB) exceeds the 5 MB limit.`);
   261	        setSelectedDocFile(null);
   262	        return;
   263	      }
   264	
   265	      setSelectedDocFile(file);
   266	    }
   267	  };
   268	
   269	  const handleUploadProjectDoc = async () => {
   270	    setErrorMsg(null);
   271	    if (!docForm.title.trim()) {
   272	      setErrorMsg('Please enter a Document / Project Title.');
   273	      return;
   274	    }
   275	    if (!docForm.description.trim()) {
   276	      setErrorMsg('Please describe what you are uploading (e.g. Prompt document, architecture notes).');
   277	      return;
   278	    }
   279	    if (!selectedDocFile) {
   280	      setErrorMsg('Please select a file to upload (.docx, .doc, .pdf, or .txt).');
   281	      return;
   282	    }
   283	
   284	    setSubmitting(true);
   285	    const { data: { session } } = await supabase.auth.getSession();
   286	    if (!session) {
   287	      setSubmitting(false);
   288	      return;
   289	    }
   290	
   291	    const studentId = session.user.id;
   292	    const now = new Date().toISOString();
   293	    const docId = generateUUID();
   294	    const cleanFileName = selectedDocFile.name.replace(/[^a-zA-Z0-9.-]/g, '_');
   295	    const storagePath = `project_docs/${studentId}/${Date.now()}_${cleanFileName}`;
   296	
   297	    try {
   298	      const { error: uploadError } = await supabase.storage
   299	        .from('student-documents')
   300	        .upload(storagePath, selectedDocFile, {
   301	          cacheControl: '3600',
   302	          upsert: true,
   303	        });
   304	
   305	      if (uploadError) {
   306	        throw new Error(`Storage upload failed: ${uploadError.message}`);
   307	      }
   308	
   309	      const newDocPayload: StudentProjectDocument = {
   310	        id: docId,
   311	        user_id: studentId,
   312	        title: docForm.title.trim(),
   313	        description: docForm.description.trim(),
   314	        file_path: storagePath,
   315	        file_name: selectedDocFile.name,
   316	        file_type: selectedDocFile.type || 'application/octet-stream',
   317	        file_size: selectedDocFile.size,
   318	        created_at: now,
   319	      };
   320	
   321	      const { data, error: dbError } = await supabase
   322	        .from('student_project_documents')
   323	        .insert([newDocPayload])
   324	        .select()
   325	        .single();
   326	
   327	      if (dbError) {
   328	        if (dbError.code === '42P01') {
   329	          console.warn("Table 'student_project_documents' pending creation in Supabase.");
   330	        } else {
   331	          throw dbError;
   332	        }
   333	      }
   334	
   335	      setProjectDocs([data || newDocPayload, ...projectDocs]);
   336	      setShowDocModal(false);
   337	      setDocForm({ title: '', description: '' });
   338	      setSelectedDocFile(null);
   339	      if (fileInputRef.current) fileInputRef.current.value = '';
   340	      setSuccessMsg('Project document uploaded successfully!');
   341	      setTimeout(() => setSuccessMsg(null), 3000);
   342	    } catch (err: any) {
   343	      setErrorMsg(err.message || 'Failed to upload project document');
   344	    } finally {
   345	      setSubmitting(false);
   346	    }
   347	  };
   348	
   349	  const handleDeleteProjectDoc = async (id: string, filePath: string) => {
   350	    if (!window.confirm('Are you sure you want to delete this project document?')) return;
   351	    try {
   352	      if (filePath) {
   353	        const cleanPath = extractStoragePath(filePath);
   354	        if (cleanPath && !cleanPath.startsWith('http')) {
   355	          await supabase.storage.from('student-documents').remove([cleanPath]);
   356	        }
   357	      }
   358	
   359	      await supabase.from('student_project_documents').delete().eq('id', id);
   360	      setProjectDocs(projectDocs.filter((d) => d.id !== id));
   361	      setSuccessMsg('Document deleted.');
   362	      setTimeout(() => setSuccessMsg(null), 3000);
   363	    } catch (err: any) {
   364	      console.warn('Error deleting doc:', err);
   365	    }
   366	  };
   367	
   368	  const handleSaveLearningNote = async () => {
   369	    setErrorMsg(null);
   370	    if (!learningForm.title.trim() || !learningForm.notes.trim()) {
   371	      setErrorMsg('Please enter a Reflection Title and Learning Process Notes.');
   372	      return;
   373	    }
   374	
   375	    setSubmitting(true);
   376	    const { data: { session } } = await supabase.auth.getSession();
   377	    if (!session) return;
   378	
   379	    try {
   380	      const newLearningPayload = {
   381	        user_id: session.user.id,
   382	        title: learningForm.title.trim(),
   383	        notes: learningForm.notes.trim(),
   384	        challenges: learningForm.challenges.trim(),
   385	        key_milestones: learningForm.key_milestones.trim(),
   386	      };
   387	
   388	      const { data, error } = await supabase
   389	        .from('student_learning_process_notes')
   390	        .insert([newLearningPayload])
   391	        .select()
   392	        .single();
   393	
   394	      if (error) {
   395	        if (error.code === '42P01') {
   396	          console.warn("Table 'student_learning_process_notes' pending creation in Supabase.");
   397	        } else {
   398	          throw error;
   399	        }
   400	      }
   401	
   402	      const savedItem: StudentLearningProcessNote = data || {
   403	        id: `learn-${Date.now()}`,
   404	        ...newLearningPayload,
   405	        created_at: new Date().toISOString(),
   406	      };
   407	
   408	      setLearningNotes([savedItem, ...learningNotes]);
   409	      setShowLearningModal(false);
   410	      setLearningForm({ title: '', notes: '', challenges: '', key_milestones: '' });
   411	      setSuccessMsg('Learning process reflection saved!');
   412	      setTimeout(() => setSuccessMsg(null), 3000);
   413	    } catch (err: any) {
   414	      setErrorMsg(err.message || 'Failed to save learning process reflection');
   415	    } finally {
   416	      setSubmitting(false);
   417	    }
   418	  };
   419	
   420	  const handleDeleteLearningNote = async (id: string) => {
   421	    if (!window.confirm('Delete this learning reflection?')) return;
   422	    try {
   423	      await supabase.from('student_learning_process_notes').delete().eq('id', id);
   424	      setLearningNotes(learningNotes.filter((n) => n.id !== id));
   425	    } catch (err: any) {
   426	      console.warn('Error deleting note:', err);
   427	    }
   428	  };
   429	
   430	  const formatFileSize = (bytes?: number) => {
   431	    if (!bytes) return '';
   432	    if (bytes < 1024) return `${bytes} B`;
   433	    if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`;
   434	    return `${(bytes / (1024 * 1024)).toFixed(1)} MB`;
   435	  };
   436	
   437	  const getDocBadge = (fileName: string) => {
   438	    const lower = fileName.toLowerCase();
   439	    if (lower.endsWith('.docx') || lower.endsWith('.doc')) {
   440	      return { label: 'Word Document', color: 'bg-blue-100 text-blue-800 border-blue-200' };
   441	    }
   442	    if (lower.endsWith('.pdf')) {
   443	      return { label: 'PDF Document', color: 'bg-red-100 text-red-800 border-red-200' };
   444	    }
   445	    if (lower.endsWith('.txt') || lower.endsWith('.md')) {
   446	      return { label: 'Prompt / Text Doc', color: 'bg-amber-100 text-amber-900 border-amber-200' };
   447	    }
   448	    return { label: 'Document', color: 'bg-blue-100 text-[#1E3A8A] border-blue-200' };
   449	  };
   450	
   451	  return (
   452	    <div className="space-y-8 animate-in fade-in duration-500">
   453	      {/* Top Banner */}
   454	      <div className="bg-gradient-to-r from-[#1E3A8A] via-[#2A4993] to-[#C49A3A] rounded-[24px] p-8 sm:p-10 text-white relative overflow-hidden shadow-lg">
   455	        <div className="absolute inset-0 bg-[url('https://www.transparenttextures.com/patterns/cubes.png')] opacity-10 mix-blend-overlay"></div>
   456	        <div className="relative z-10 max-w-3xl">
   457	          <div className="inline-flex items-center gap-2 px-3 py-1 rounded-full bg-white/15 backdrop-blur-md text-xs font-semibold uppercase tracking-wider mb-3">
   458	            <Sparkles className="w-3.5 h-3.5 text-[#E6C975]" />
   459	            Continuous Growth & Reflections
   460	          </div>
   461	          <h2 className="text-3xl sm:text-4xl font-serif font-bold mb-3 tracking-tight">Logs & Reflections</h2>
   462	          <p className="text-white/90 text-sm sm:text-base leading-relaxed">
   463	            Record meeting discussions, log newly mastered AI technologies, upload project & prompt documents (.docx, .pdf, .doc), and reflect on your overall learning methodology.
   464	          </p>
   465	        </div>
   466	      </div>
   467	
   468	      {/* Success Notification */}
   469	      {successMsg && (
   470	        <motion.div
   471	          initial={{ opacity: 0, y: -10 }}
   472	          animate={{ opacity: 1, y: 0 }}
   473	          className="p-4 bg-emerald-50 rounded-2xl flex items-center gap-3 border border-emerald-200 text-emerald-800 text-sm font-semibold shadow-xs"
   474	        >
   475	          <CheckCircle2 className="w-5 h-5 text-emerald-600 flex-shrink-0" />
   476	          <span>{successMsg}</span>
   477	        </motion.div>
   478	      )}
   479	
   480	      {/* Section Quick Navigation Pills */}
   481	      <div className="bg-white rounded-2xl border border-[#E8DED0] p-4 shadow-sm flex flex-wrap items-center justify-between gap-3">
   482	        <div className="flex flex-wrap items-center gap-2">
   483	          <button type="button"
   484	            onClick={() => setActiveSectionView('all')}
   485	            className={`px-4 py-2 rounded-xl text-xs sm:text-sm font-bold transition-all cursor-pointer ${
   486	              activeSectionView === 'all'
   487	                ? 'bg-[#1E3A8A] text-white shadow-sm'
   488	                : 'bg-[#F8F5F0] text-[#6B5A4D] hover:bg-[#EFE9DF]'
   489	            }`}
   490	          >
   491	            All Sections ({meetingNotes.length + skillUpdates.length + projectDocs.length + learningNotes.length})
   492	          </button>
   493	          <button type="button"
   494	            onClick={() => setActiveSectionView('notes')}
   495	            className={`px-4 py-2 rounded-xl text-xs sm:text-sm font-bold transition-all cursor-pointer flex items-center gap-1.5 ${
   496	              activeSectionView === 'notes'
   497	                ? 'bg-[#1E3A8A] text-white shadow-sm'
   498	                : 'bg-[#F8F5F0] text-[#6B5A4D] hover:bg-[#EFE9DF]'
   499	            }`}
   500	          >
   501	            <BookOpen className="w-4 h-4 text-[#C49A3A]" />
   502	            Meeting Notes ({meetingNotes.length})
   503	          </button>
   504	          <button type="button"
   505	            onClick={() => setActiveSectionView('skills')}
   506	            className={`px-4 py-2 rounded-xl text-xs sm:text-sm font-bold transition-all cursor-pointer flex items-center gap-1.5 ${
   507	              activeSectionView === 'skills'
   508	                ? 'bg-[#1E3A8A] text-white shadow-sm'
   509	                : 'bg-[#F8F5F0] text-[#6B5A4D] hover:bg-[#EFE9DF]'
   510	            }`}
   511	          >
   512	            <Sparkles className="w-4 h-4 text-emerald-600" />
   513	            Skills & Tech ({skillUpdates.length})
   514	          </button>
   515	          <button type="button"
   516	            onClick={() => setActiveSectionView('project-docs')}
   517	            className={`px-4 py-2 rounded-xl text-xs sm:text-sm font-bold transition-all cursor-pointer flex items-center gap-1.5 ${
   518	              activeSectionView === 'project-docs'
   519	                ? 'bg-[#1E3A8A] text-white shadow-sm'
   520	                : 'bg-[#F8F5F0] text-[#6B5A4D] hover:bg-[#EFE9DF]'
   521	            }`}
   522	          >
   523	            <FileCode className="w-4 h-4 text-blue-600" />
   524	            Project & Prompt Docs ({projectDocs.length})
   525	          </button>
   526	          <button type="button"
   527	            onClick={() => setActiveSectionView('learning-process')}
   528	            className={`px-4 py-2 rounded-xl text-xs sm:text-sm font-bold transition-all cursor-pointer flex items-center gap-1.5 ${
   529	              activeSectionView === 'learning-process'
   530	                ? 'bg-[#1E3A8A] text-white shadow-sm'
   531	                : 'bg-[#F8F5F0] text-[#6B5A4D] hover:bg-[#EFE9DF]'
   532	            }`}
   533	          >
   534	            <Compass className="w-4 h-4 text-[#1E3A8A]" />
   535	            Learning Process ({learningNotes.length})
   536	          </button>
   537	        </div>
